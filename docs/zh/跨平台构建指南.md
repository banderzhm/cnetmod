# 跨平台构建指南

本指南介绍如何在不同操作系统和编译器环境下构建 cnetmod HTTP/3 模块。QUIC 实现依赖 BoringSSL，因此构建配置需要特别注意依赖安装。

---

## Windows (MSVC + vcpkg + BoringSSL)

### 前置依赖安装

**1. 安装 Visual Studio 2022 Build Tools**

下载并安装 [Visual Studio Build Tools 2022](https://visualstudio.microsoft.com/downloads/)，勾选以下组件：
- ✓ C++ 生成工具
- ✓ C++23 模块实验性支持（MSVC v143）
- ✓ Windows 11 SDK
- ✓ CMake tools for Windows

**2. 安装 vcpkg**

```powershell
# 克隆 vcpkg 到工作目录
cd C:\dev
git clone https://github.com/Microsoft/vcpkg.git
cd vcpkg

# 执行 bootstrap
.\bootstrap-vcpkg.bat

# 添加到 PATH（可选，推荐）
$env:Path += ";C:\dev\vcpkg"
[System.Environment]::SetEnvironmentVariable("Path", $env:Path, "User")
```

**3. 安装 BoringSSL**

```powershell
# 使用 vcpkg 安装 BoringSSL（包含 QUIC API 支持）
.\vcpkg install boringssl:x64-windows

# 验证安装
.\vcpkg list | findstr boringssl
# 输出示例：boringssl:x64-windows  2024.x.x

# 可选：安装静态库版本（推荐用于发布）
.\vcpkg install boringssl:x64-windows-static
```

**4. 配置自动集成**

```powershell
# 全局集成（影响所有 CMake 项目）
.\vcpkg integrate install

# 或项目级集成（通过 CMake 工具链文件）
# cmake -DCMAKE_TOOLCHAIN_FILE=C:/dev/vcpkg/scripts/buildsystems/vcpkg.cmake
```

### CMake 构建命令

**Debug 构建**

```powershell
# 生成构建系统
cmake -G "Visual Studio 17 2022" `
      -A x64 `
      -DCMAKE_BUILD_TYPE=Debug `
      -DCMAKE_TOOLCHAIN_FILE="C:/dev/vcpkg/scripts/buildsystems/vcpkg.cmake" `
      -DCNETMOD_ENABLE_QUIC=ON `
      -DCNETMOD_ENABLE_BORINGSSL_QUIC=ON `
      -S E:\github\cnetmod -B cmake-build-debug

# 编译
cmake --build cmake-build-debug --config Debug

# 运行测试
ctest --test-dir cmake-build-debug -C Debug -R quic_
```

**Release 构建**

```powershell
# 生成构建系统（启用 LTO 优化）
cmake -G "Visual Studio 17 2022" `
      -A x64 `
      -DCMAKE_BUILD_TYPE=Release `
      -DCMAKE_TOOLCHAIN_FILE="C:/dev/vcpkg/scripts/buildsystems/vcpkg.cmake" `
      -DCNETMOD_ENABLE_QUIC=ON `
      -DCNETMOD_ENABLE_BORINGSSL_QUIC=ON `
      -DCNETMOD_ENABLE_LTO=ON `
      -S E:\github\cnetmod -B cmake-build-release

# 编译（使用所有 CPU 核心）
cmake --build cmake-build-release --config Release --parallel

# 运行测试
ctest --test-dir cmake-build-release -C Release -R quic_ --output-on-failure

# 安装（可选）
cmake --install cmake-build-release --prefix C:\dev\cnetmod-install
```

### MSVC 模块扫描配置

**编译器标志说明**：

```cmake
# CMakeLists.txt 中的 MSVC 特定配置
if(MSVC)
    # C++23 标准
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    
    # 启用 C++23 模块支持
    add_compile_options(
        /experimental:cxxmodules    # 模块扫描
        /std:c++latest              # C++23 标准
        /Zc:__cplusplus             # 正确的 __cplusplus 宏值
        /EHsc                       # C++ 异常处理
        /permissive-                # 严格符合标准
    )
    
    # 模块生成模式（MSVC 17.8+）
    set(CMAKE_CXX_MODULE_GENERATION_MODE "SEPARATE")
endif()
```

**手动指定模块文件**：

```cmake
# 如果 CMake 无法自动扫描模块依赖
add_library(cnetmod_http3)

target_sources(cnetmod_http3
    PUBLIC
        FILE_SET CXX_MODULES
        FILES
            src/http/v3/http3_server.cppm
            src/http/v3/http3_client.cppm
            src/http/v3/http3_frame.cppm
            src/http/v3/qpack_encoder.cppm
            src/http/v3/qpack_decoder.cppm
)
```

## Linux (Clang + Ninja + libssl-dev)

### Ubuntu/Debian 前置安装

```bash
# 更新包索引
sudo apt update

# 安装构建工具
sudo apt install -y \
    clang \
    clang-format \
    ninja-build \
    cmake \
    pkg-config \
    git \
    curl

# 安装 OpenSSL（基础依赖，非 QUIC）
sudo apt install -y libssl-dev

# 验证版本
clang --version    # 要求 18+
cmake --version    # 要求 3.25+
ninja --version    # 要求 1.11+
```

### Arch Linux

```bash
# 更新系统
sudo pacman -Syu

# 安装构建工具
sudo pacman -S \
    clang \
    ninja \
    cmake \
    openssl \
    base-devel \
    git

# 验证
clang --version
cmake --version
```

### 构建命令

**Clang 18+ 推荐配置**

```bash
# 配置 CMake（Release 模式）
/usr/bin/cmake -GNinja \
       -DCMAKE_BUILD_TYPE=Release \
       -DCMAKE_C_COMPILER=/usr/bin/clang \
       -DCMAKE_CXX_COMPILER=/usr/bin/clang++ \
       -DCMAKE_CXX_FLAGS="-fmodules -stdlib=libc++" \
       -DCNETMOD_ENABLE_QUIC=ON \
       -DCNETMOD_ENABLE_SSL=ON \
       -S . -B build

# 编译（使用所有核心）
cmake --build build -j$(nproc)

# 运行测试
ctest --test-dir build -R test_quic --output-on-failure

# 安装到系统
sudo cmake --install build --prefix /usr/local
```

**Debug 构建**

```bash
cmake -GNinja \
      -DCMAKE_BUILD_TYPE=Debug \
      -DCMAKE_C_COMPILER=/usr/bin/clang \
      -DCMAKE_CXX_COMPILER=/usr/bin/clang++ \
      -DCMAKE_CXX_FLAGS="-fmodules -stdlib=libc++ -fsanitize=address,undefined" \
      -DCNETMOD_ENABLE_QUIC=ON \
      -DCNETMOD_ENABLE_SSL=ON \
      -S . -B build-debug

# 编译并运行
cmake --build build-debug -j$(nproc)
ctest --test-dir build-debug -R quic_ --output-on-failure
```

### BoringSSL 手动安装

**原因**：Ubuntu apt 版本的 OpenSSL 不支持 QUIC API，必须手动编译 BoringSSL。

```bash
# 克隆 BoringSSL
cd /tmp
git clone https://boringssl.googlesource.com/boringssl
cd boringssl

# 配置（禁用测试和基准）
cmake -GNinja \
      -DCMAKE_BUILD_TYPE=Release \
      -DCMAKE_C_COMPILER=/usr/bin/clang \
      -DCMAKE_CXX_COMPILER=/usr/bin/clang++ \
      -DCMAKE_POSITION_INDEPENDENT_CODE=ON \
      -S . -B build

# 编译
cmake --build build -j$(nproc)

# 安装到系统
sudo mkdir -p /usr/local/include
sudo cp -r include/openssl /usr/local/include/
sudo cp build/libcrypto.a build/libssl.a /usr/local/lib/

# 验证安装
ls -l /usr/local/lib/libcrypto.a
ls -l /usr/local/include/openssl/ssl.h

# 更新 ldconfig
sudo ldconfig
```

**在 cnetmod 中使用手动安装的 BoringSSL**

```bash
# 指定 BoringSSL 路径
cmake -GNinja \
      -DCMAKE_BUILD_TYPE=Release \
      -DBORINGSSL_ROOT=/usr/local \
      -DBORINGSSL_INCLUDE_DIR=/usr/local/include \
      -DBORINGSSL_LIBRARY=/usr/local/lib/libssl.a \
      -DBORINGSSL_CRYPTO_LIBRARY=/usr/local/lib/libcrypto.a \
      -DCNETMOD_ENABLE_QUIC=ON \
      -DCNETMOD_ENABLE_BORINGSSL_QUIC=ON \
      -S . -B build
```

## WSL2 环境特殊注意事项

### PATH 问题

```powershell
# PowerShell 中设置 WSL 二进制路径
$env:CNETMOD_WSL_BIN = "/mnt/e/github/cnetmod/cmake-build-release-wsl"

# 或在 WSL 中直接设置
export PATH="/mnt/e/github/cnetmod/cmake-build-release-wsl:$PATH"
```

### 文件共享访问

**性能警告**：

```bash
# ❌ 不推荐：通过 /mnt/c/... 访问 Windows 文件
cd /mnt/c/Users/dev/projects/cnetmod
cmake --build build  # 非常慢（跨文件系统 I/O）

# ✓ 推荐：在 Linux 文件系统内克隆
cd ~
git clone https://github.com/example/cnetmod.git
cd cnetmod
cmake --build build  # 快速（原生 ext4）
```

**Git Worktree 方案**（同时访问 Windows 和 Linux 源码）

```bash
# Windows 侧（PowerShell）
cd E:\github\cnetmod
git worktree add ../cnetmod-wsl HEAD

# WSL 侧
cd /mnt/e/github/cnetmod-wsl
cmake --build build  # 使用 worktree 避免复制
```

### 双重构建配置

**推荐分离策略**：

| 环境 | 用途 | 构建命令 |
|------|------|---------|
| WSL | 编译 + 单元测试 | `cmake --build build -j$(nproc)` |
| Windows | IDE 调试 + 集成测试 | VS 打开 `cmake-build-debug` |
| Docker | 生产构建 | `docker build -t cnetmod .` |

**共享构建缓存**

```bash
# 使用 ccache 加速重复编译
sudo apt install ccache
export CC="ccache clang"
export CXX="ccache clang++"

# 缓存大小限制
ccache -M 5G
```

## Docker 构建镜像方案

### 基础构建镜像

**Dockerfile**：

```dockerfile
FROM ubuntu:22.04 as builder

# 避免交互式安装提示
ENV DEBIAN_FRONTEND=noninteractive

# 安装依赖
RUN apt-get update && apt-get install -y \
    clang-14 \
    clang-format-14 \
    ninja-build \
    cmake \
    pkg-config \
    python3 \
    python3-pip \
    git \
    curl \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# 设置默认编译器
ENV CC=/usr/bin/clang-14
ENV CXX=/usr/bin/clang++-14

# 克隆 BoringSSL（可选，如需 QUIC 支持）
RUN git clone https://boringssl.googlesource.com/boringssl /opt/boringssl \
    && cd /opt/boringssl \
    && cmake -GNinja -DCMAKE_BUILD_TYPE=Release -S . -B build \
    && cmake --build build -j$(nproc) \
    && cp -r include/openssl /usr/local/include/ \
    && cp build/libcrypto.a build/libssl.a /usr/local/lib/

# 复制源码
WORKDIR /workspace
COPY . .

# 配置 CMake
RUN /usr/bin/cmake -GNinja \
    -DCMAKE_BUILD_TYPE=Release \
    -DCMAKE_C_COMPILER=/usr/bin/clang-14 \
    -DCMAKE_CXX_COMPILER=/usr/bin/clang++-14 \
    -DCNETMOD_ENABLE_QUIC=ON \
    -DCNETMOD_ENABLE_BORINGSSL_QUIC=ON \
    -S /workspace -B /workspace/build

# 编译
RUN cmake --build /workspace/build -j$(nproc)

# 运行测试
CMD ["ctest", "--test-dir", "/workspace/build", "--output-on-failure"]
```

**构建和运行**：

```bash
# 构建镜像
docker build -t cnetmod-builder .

# 运行测试
docker run --rm cnetmod-builder

# 进入容器调试
docker run --rm -it cnetmod-builder /bin/bash
```

### 多阶段构建（生产部署）

```dockerfile
# 阶段 1: 构建
FROM ubuntu:22.04 as builder
# ... (同上构建步骤)

# 阶段 2: 运行时（最小镜像）
FROM ubuntu:22.04

RUN apt-get update && apt-get install -y \
    libssl3 \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# 复制编译产物
COPY --from=builder /workspace/build/cnetmod_app /usr/local/bin/
COPY --from=builder /workspace/config /etc/cnetmod/

EXPOSE 4433/udp

ENTRYPOINT ["/usr/local/bin/cnetmod_app"]
CMD ["--config", "/etc/cnetmod/server.yaml"]
```

## 常见问题排查

### Q: "module not found" 编译错误

**原因**: MSVC 无法识别 `.cppm` 文件扩展名或模块扫描失败。

**解决**：

```cmake
# 显式启用模块支持
if(MSVC)
    add_compile_options(
        /Zc:__cplusplus            # 正确的 __cplusplus 宏
        /experimental:cxxmodules   # 模块扫描
        /std:c++latest             # C++23 标准
    )
endif()

# 或手动指定模块文件
target_sources(my_target
    PUBLIC
        FILE_SET CXX_MODULES
        FILES src/module.cppm
)
```

### Q: "BoringSSL not found"

**原因**: vcpkg 安装的 BoringSSL 路径未在 CMake 搜索路径中。

**解决**：

```bash
# 显式指定 BORINGSSL_ROOT
cmake -DBORINGSSL_ROOT="C:/dev/vcpkg/installed/x64-windows" \
      -DCMAKE_PREFIX_PATH="C:/dev/vcpkg/installed/x64-windows" \
      ...

# 或使用 vcpkg 工具链文件
cmake -DCMAKE_TOOLCHAIN_FILE=C:/dev/vcpkg/scripts/buildsystems/vcpkg.cmake \
      ...
```

### Q: "udp socket fails on Windows"

**原因**: Windows 防火墙阻止 UDP 端口或需要管理员权限。

**解决**：

```powershell
# 方案 1：添加防火墙例外（推荐）
netsh advfirewall firewall add rule name="cnetmod HTTP3" `
    dir=in action=allow protocol=UDP localport=4433

# 方案 2：以管理员身份运行程序

# 方案 3：临时禁用防火墙（仅用于测试）
netsh advfirewall set allprofiles state off
# ⚠️ 测试完成后立即恢复
netsh advfirewall set allprofiles state on
```

### Q: "Clang modules not supported"

**原因**: Clang 版本过低或不支持 `-fmodules` 标志。

**解决**：

```bash
# 升级 Clang 到 18+
wget https://apt.llvm.org/llvm.sh
chmod +x llvm.sh
sudo ./llvm.sh 18

# 使用新编译器
export CC=/usr/bin/clang-18
export CXX=/usr/bin/clang++-18
```

## 性能优化建议

### Windows 优化

```cmake
# 启用 GSO/GRO（Generic Segmentation/Receive Offload）
option(ENABLE_GSO "Enable Generic Segmentation Offload" ON)

# 启用 IOCP 深度集成
option(ENABLE_IOCP "Enable I/O Completion Ports" ON)

# Release 构建优化标志
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} /O2 /GL")
```

### Linux 优化

```cmake
# io_uring 模式优先（内核 ≥ 5.6）
option(ENABLE_IO_URING "Enable io_uring backend" ON)

# 启用 eBPF XDP（可选，高性能场景）
option(ENABLE_XDP "Enable eXpress Data Path" OFF)

# Release 优化
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3 -flto")
```

### 通用优化

```cmake
# Link-Time Optimization（LTO）
option(CNETMOD_ENABLE_LTO "Enable Link-Time Optimization" ON)

# Profile-Guided Optimization（PGO）
# 步骤 1：生成 profile
cmake -DCMAKE_CXX_FLAGS="-fprofile-generate" ...
# 运行负载测试
# 步骤 2：使用 profile 重新编译
cmake -DCMAKE_CXX_FLAGS="-fprofile-use=/path/to/profile" ...
```

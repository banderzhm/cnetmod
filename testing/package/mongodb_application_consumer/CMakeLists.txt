cmake_minimum_required(VERSION 4.0)
project(cnetmod_mongodb_application_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
find_package(cnetmod 2 CONFIG REQUIRED COMPONENTS application)
enable_testing()
foreach(name IN ITEMS credentials application)
    add_executable(cnetmod_mongodb_${name}_consumer ${name}.cpp)
    target_link_libraries(cnetmod_mongodb_${name}_consumer PRIVATE cnetmod::application)
    add_test(NAME installed_mongodb_${name}_consumer COMMAND cnetmod_mongodb_${name}_consumer)
endforeach()

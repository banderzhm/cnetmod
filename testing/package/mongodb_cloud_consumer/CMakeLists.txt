cmake_minimum_required(VERSION 4.0)
project(cnetmod_mongodb_cloud_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
find_package(cnetmod 2 CONFIG REQUIRED COMPONENTS mongodb_cloud)
add_executable(cnetmod_mongodb_cloud_consumer main.cpp)
target_link_libraries(cnetmod_mongodb_cloud_consumer PRIVATE cnetmod::mongodb_cloud)
enable_testing()
add_test(NAME installed_mongodb_cloud_consumer COMMAND cnetmod_mongodb_cloud_consumer)
find_package(cnetmod 2 CONFIG REQUIRED COMPONENTS application)
add_executable(cnetmod_mongodb_application_consumer application.cpp)
target_link_libraries(cnetmod_mongodb_application_consumer PRIVATE cnetmod::application)
add_test(NAME installed_mongodb_application_consumer COMMAND cnetmod_mongodb_application_consumer)
